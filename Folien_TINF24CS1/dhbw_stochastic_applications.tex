\begin{frame}
    \frametitle{Diskrete Wahrscheinlichkeitsverteilung}
\framesubtitle{}

\begin{block}{Satz der totalen Wahrscheinlichkeit}
Für eine Zerlegung  $\Omega = \bigcup_{j=1}^{n} B_j, \text{ mit } B_i \cap B_k = \emptyset \text{ für } i \neq k $
\begin{align*}
& P(A ) = \sum_{j=1}^{n}  P(A \; | \;  B_j) \cdot P(B_j)
\end{align*}
\end{block}

\begin{block}{Beweis}
Da $A = A \cap \Omega = A \cap \bigcup_{j=1}^{n} B_j = \bigcup_{j=1}^{n} (A \cap B_j)$ eine disjunkte Zerlegung ist, folgt:
\begin{align*}
P(A) &= \sum_{j=1}^{n} P(A \cap B_j) = \sum_{j=1}^{n} P(A \; | \; B_j) \cdot P(B_j)
\end{align*}
\end{block}

 \end{frame}



\begin{frame}
    \frametitle{Diskrete Wahrscheinlichkeitsverteilung}
\framesubtitle{}

\begin{block}{Satz von Bayes}
Für $A,B \in \mathcal{A}$ mit  $P(B) > 0$ gilt
\begin{align*}
& P(A \; | \;  B) = \frac{P(B \; | \; A) \cdot P(A)} {P(B)} \\
\end{align*}
\end{block}

\begin{block}{Beweis}
\begin{align*}
& P(A \; | \;  B) =\frac{P(A \cap B)}{P(B)} = \frac{ \frac{P(A \cap B) \cdot P(A)}{P(A)}}{P(B)}  =  \frac{P(B \; | \; A) \cdot P(A)} {P(B)} 
\end{align*}
\end{block}



 \end{frame}


\begin{frame}
    \frametitle{Diskrete Wahrscheinlichkeitsverteilung}
\framesubtitle{}

\begin{block}{Das Ziegenproblem (Monty Hall)}
Es gibt drei Türen. Hinter genau einer steht das Auto, hinter den beiden anderen je eine Ziege.
Der Spieler wählt zunächst irgendeine der drei Türen.
Danach öffnet der Moderator, der die Lage des Autos kennt, immer eine andere Tür mit Ziege.
Anschließend darf der Spieler zur einzig verbleibenden geschlossenen Tür wechseln.
\end{block}

 \end{frame}


\begin{frame}
    \frametitle{Diskrete Wahrscheinlichkeitsverteilung}
\framesubtitle{}

\begin{block}{Abstrakte Ereignisse}
Wir definieren
\begin{align*}
&A = \{\text{die erste Wahl des Spielers ist richtig}\},\\
&W = \{\text{der Spieler gewinnt beim Wechseln}\}.
\end{align*}
Da genau eine von drei Türen das Auto enthält, gilt sofort
\begin{align*}
P(A)=\frac13.
\end{align*}
\end{block}

 \end{frame}


\begin{frame}
    \frametitle{Diskrete Wahrscheinlichkeitsverteilung}
\framesubtitle{}

\begin{block}{Rigoroser Beweis}
Aus der Spielregel folgt unmittelbar
\begin{align*}
P(W \mid A)=0,
\qquad
P(W \mid A^c)=1.
\end{align*}
Denn im Fall \(A\) ist die erste Wahl bereits richtig, also verliert Wechseln.
Im Fall \(A^c\) ist die erste Wahl falsch; der Moderator muss dann die einzige Ziegentür unter den beiden anderen öffnen, und die verbleibende geschlossene Tür enthält das Auto.
\end{block}

 \end{frame}


\begin{frame}
    \frametitle{Diskrete Wahrscheinlichkeitsverteilung}
\framesubtitle{}

\begin{block}{Schlussfolgerung mit bedingter Wahrscheinlichkeit}
Mit dem Satz der totalen Wahrscheinlichkeit folgt
\begin{align*}
P(W)
&= P(W \mid A)P(A) + P(W \mid A^c)P(A^c) \\
&= 0 \cdot \frac13 + 1 \cdot \frac23
= \frac23.
\end{align*}
Die Gewinnchance beim Bleiben ist dagegen
\begin{align*}
P(A)=\frac13.
\end{align*}
Also ist Wechseln optimal. Die genaue Auswahlregel des Moderators unter mehreren möglichen Ziegentüren ist dabei egal, solange er immer eine Ziegentür öffnet und Wechseln anbietet.
\end{block}

 \end{frame}


\begin{frame}
    \frametitle{Diskrete Wahrscheinlichkeitsverteilung}
\framesubtitle{}

\begin{block}{Bayes'sche Inferenz (Bayesian Inference)}
Die Formel von Bayes beschreibt, wie wir eine Hypothese $H$ nach Beobachtung von Daten $D$ aktualisieren:
\begin{align*}
P(H \mid D) = \frac{P(D \mid H) \cdot P(H)}{P(D)}
\end{align*}
Dabei heißen
\begin{align*}
&P(H) &&\text{Prior}, \\
&P(D \mid H) &&\text{Likelihood}, \\
&P(H \mid D) &&\text{Posterior}.
\end{align*}
\end{block}

 \end{frame}


\begin{frame}[shrink=5]
    \frametitle{Diskrete Wahrscheinlichkeitsverteilung}
\framesubtitle{}

\begin{block}{Interpretation der Bayes'schen Inferenz}
Wir starten mit einer anfänglichen Einschätzung $P(H)$ über eine Hypothese und passen diese durch neue Daten $D$ an.
Ist $P(D \mid H)$ groß, dann sprechen die Daten für die Hypothese; ist $P(D \mid H)$ klein, dann spricht das gegen sie.
\end{block}

\begin{block}{Prior, Likelihood, Posterior}
\begin{align*}
&\text{Prior } P(H): &&\text{unsere Einschätzung vor den Daten}, \\
&\text{Likelihood } P(D \mid H): &&\text{wie gut erklärt } H \text{ die Daten } D, \\
&\text{Posterior } P(H \mid D): &&\text{unsere Einschätzung nach den Daten}.
\end{align*}
Wichtig: Die Likelihood ist nicht $P(H \mid D)$, sondern $P(D \mid H)$. Sie bewertet also die Daten unter der Annahme, dass die Hypothese gilt.
\end{block}

\begin{block}{Typischer Ablauf}
\begin{enumerate}
\item Formuliere mögliche Hypothesen $H_1,\ldots,H_k$.
\item Lege Prior-Wahrscheinlichkeiten $P(H_i)$ fest.
\item Modelliere die Datenwahrscheinlichkeit $P(D \mid H_i)$.
\item Berechne daraus die Posterior-Wahrscheinlichkeiten $P(H_i \mid D)$.
\item Entscheide z.B. mit der wahrscheinlichsten Hypothese (MAP-Regel).
\end{enumerate}
\end{block}

 \end{frame}


\begin{frame}
    \frametitle{Diskrete Wahrscheinlichkeitsverteilung}
\framesubtitle{}

\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{Bayesianische Sicht}
Wahrscheinlichkeit misst Unsicherheit.
Darum dürfen auch Hypothesen oder Parameter Wahrscheinlichkeiten tragen:
\begin{align*}
P(H), \qquad P(H \mid D), \\
p(\theta \mid x).
\end{align*}
Neue Daten aktualisieren dieses Wissen durch Bayes.
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{Frequentistische Sicht}
Wahrscheinlichkeit beschreibt relative Häufigkeiten.
Ein Parameter \(\theta\) ist fest; zufällig sind Daten, Schätzer, Tests und Intervalle.
\end{block}
\end{column}
\end{columns}

 \end{frame}


\begin{frame}
    \frametitle{Diskrete Wahrscheinlichkeitsverteilung}
\framesubtitle{}

\begin{block}{Wissenschaftsphilosophische Differenz}
Bayesianer fragen: ``Wie plausibel ist eine Hypothese nach den Daten?''
Frequentisten fragen: ``Wie verhält sich ein Verfahren bei vielen Wiederholungen?''
\end{block}

\begin{block}{Was ist zufällig?}
Bayesianisch: Auch Hypothesen dürfen Wahrscheinlichkeiten tragen, zum Beispiel
\[
P(H \mid D).
\]
Frequentistisch: Eine Hypothese ist nicht zufällig, sondern entweder wahr oder falsch; zufällig sind nur die Daten und damit das statistische Verfahren.
\end{block}

 \end{frame}


\begin{frame}
    \frametitle{Diskrete Wahrscheinlichkeitsverteilung}
\framesubtitle{}

\begin{block}{Was bedeuten $95\%$?}
Bayesianisch kann man sinnvoll sagen:
\[
P(H \mid D)=0.95.
\]
Das bedeutet: Nach dem Modell und den beobachteten Daten hat die Hypothese \(H\) eine Plausibilität von $95\%$.
\end{block}

 \end{frame}


\begin{frame}
    \frametitle{Diskrete Wahrscheinlichkeitsverteilung}
\framesubtitle{}

\begin{block}{Frequentistische Deutung von $95\%$}
Frequentistisch bezieht sich $95\%$ typischerweise auf ein Verfahren, also auf eine Fehlerquote oder Trefferquote bei vielen gedachten Wiederholungen.
\end{block}

\begin{block}{Wichtige Folgerung}
Ein frequentistisches Signifikanzniveau von $5\%$ bedeutet daher nicht:
``Die Nullhypothese ist mit Wahrscheinlichkeit $5\%$ wahr.''
Gemeint ist vielmehr: Wenn die Nullhypothese wahr ist, produziert das Verfahren langfristig nur in etwa $5\%$ der Fälle einen falschen Alarm.
\end{block}

\begin{block}{Einordnung}
Bayesianische Verfahren sind stark bei explizitem Vorwissen.
Frequentistische Verfahren sind stark bei Fehlerkontrolle und Kalibrierung.
Oft sind das zwei komplementäre Perspektiven.
\end{block}

 \end{frame}


\begin{frame}
    \frametitle{Diskrete Wahrscheinlichkeitsverteilung}
\framesubtitle{}

\begin{block}{Black-Swan-Paradoxon in bayesianischer Sicht}
Sei \(H=\{\text{alle Schwäne sind weiß}\}\),
\(D_n=\{\text{die ersten } n \text{ beobachteten Schwäne sind weiß}\}\)
und \(B=\{\text{ein beobachteter Schwan ist schwarz}\}\).
Dann gilt nach Bayes
\[
P(H \mid D_n)=\frac{P(D_n \mid H)P(H)}{P(D_n)}.
\]
Unter \(H\) ist jede weiße Beobachtung sicher, also \(P(D_n \mid H)=1\).
Weiße Schwäne sprechen daher für \(H\), aber sie beweisen \(H\) nicht, weil auch andere Hypothesen viele weiße Schwäne gut erklären können.
\end{block}

 \end{frame}


\begin{frame}
    \frametitle{Diskrete Wahrscheinlichkeitsverteilung}
\framesubtitle{}

\begin{block}{Asymmetrie der Evidenz}
Beobachten wir dagegen einen schwarzen Schwan \(B\), so gilt unter \(H\):
\[
P(B \mid H)=0.
\]
Damit folgt sofort
\[
P(H \mid B)=0.
\]
Viele bestätigende Beobachtungen machen eine All-Aussage also nur schrittweise plausibler; ein echtes Gegenbeispiel zerstört sie sofort.
\end{block}

 \end{frame}


\begin{frame}
    \frametitle{Diskrete Wahrscheinlichkeitsverteilung}
\framesubtitle{}

\begin{block}{Konkretes bayesianisches Beispiel}
Vergleichen wir zwei Hypothesen mit gleichem Prior:
\[
P(H)=P(H')=\frac12,
\]
wobei
\begin{align*}
H&=\{\text{alle Schwäne sind weiß}\}, \\
H'&=\{95\% \text{ aller Schwäne sind weiß}\}.
\end{align*}
Dann gilt für \(n\) weiße Beobachtungen:
\[
P(D_n \mid H)=1,
\qquad
P(D_n \mid H')=0.95^n.
\]
\end{block}

 \end{frame}


\begin{frame}
    \frametitle{Diskrete Wahrscheinlichkeitsverteilung}
\framesubtitle{}

\begin{block}{Posterior ausrechnen}
Da wir nur die beiden Hypothesen \(H\) und \(H'\) zulassen, ist
\[
P(D_n)=P(D_n \mid H)P(H)+P(D_n \mid H')P(H').
\]
Mit Bayes folgt daher
\begin{align*}
P(H \mid D_n)
&= \frac{P(D_n \mid H)P(H)}{P(D_n \mid H)P(H)+P(D_n \mid H')P(H')} \\
&= \frac{1 \cdot \frac12}{1 \cdot \frac12 + 0.95^n \cdot \frac12}.
\end{align*}
Im Zähler und Nenner kann man den gemeinsamen Faktor \(\frac12\) kürzen. Also
\[
P(H \mid D_n)
= \frac{1}{1+0.95^n}.
\]
\end{block}

 \end{frame}


\begin{frame}
    \frametitle{Diskrete Wahrscheinlichkeitsverteilung}
\framesubtitle{}

\begin{block}{Bedeutung der Formel}
Zum Beispiel:
\[
P(H \mid D_{10}) \approx 0.625,
\qquad
P(H \mid D_{100}) \approx 0.994.
\]
Mit wachsendem \(n\) wird der Term \(0.95^n\) immer kleiner.
Damit fällt die Konkurrenzhypothese \(H'\) im Nenner immer weniger ins Gewicht.
Viele weiße Schwäne machen \(H\) also sehr plausibel, aber nicht durch logische Notwendigkeit, sondern relativ zu den zugelassenen Konkurrenzhypothesen.
\end{block}

 \end{frame}


\begin{frame}
    \frametitle{Diskrete Wahrscheinlichkeitsverteilung}
\framesubtitle{}

\begin{block}{Philosophische Pointe}
Bayesianisch ist das kein Widerspruch, sondern eine Folge von Likelihood und Prior.
Die Daten allein erzeugen keine Theorie aus dem Nichts, sondern aktualisieren nur, was vorher als möglich modelliert war.
\end{block}

\begin{block}{Induktion braucht Vorannahmen}
Wenn der Prior \(P(H)=0\) wäre, dann bliebe für alle \(n\)
\[
P(H \mid D_n)=0.
\]
Auch unendlich viele weiße Schwäne könnten \(H\) dann nie plausibel machen.
Gerade darin zeigt sich die philosophische Tiefe: Induktives Lernen funktioniert nur relativ zu einem Modellraum und zu Vorannahmen.
\end{block}

 \end{frame}


\begin{frame}[shrink=5]
    \frametitle{Diskrete Wahrscheinlichkeitsverteilung}
\framesubtitle{}

\begin{block}{Beispiel: medizinischer Test}
Sei $H=$ ``Person ist krank'' und $D=$ ``Test ist positiv''.
Angenommen:
\begin{align*}
&P(H)=0.01, \\
&P(D \mid H)=0.99, \\
&P(D \mid H^c)=0.05.
\end{align*}
Dann ist der Prior klein: Nur $1\%$ der Personen sind krank. Die Likelihood $P(D \mid H)=0.99$ sagt: Wenn jemand krank ist, ist der Test fast immer positiv.

Wichtig: $P(D \mid H)$ und $P(D \mid H^c)$ sind keine Gegenwahrscheinlichkeiten, weil unterschiedlich konditioniert wird. Komplementär wären hier zum Beispiel $P(D \mid H)$ und $P(D^c \mid H)$.
\end{block}

\begin{block}{Posterior berechnen}
Mit Bayes folgt
\begin{align*}
P(H \mid D)
&= \frac{P(D \mid H)P(H)}{P(D \mid H)P(H) + P(D \mid H^c)P(H^c)} \\
&= \frac{0.99 \cdot 0.01}{0.99 \cdot 0.01 + 0.05 \cdot 0.99}
\approx 0.167.
\end{align*}
Obwohl der Test gut ist, liegt die Wahrscheinlichkeit nach einem positiven Test also nur bei etwa $16.7\%$.
\end{block}

 \end{frame}


\begin{frame}[shrink=5]
    \frametitle{Diskrete Wahrscheinlichkeitsverteilung}
\framesubtitle{}

\begin{block}{Bayes bei kontinuierlichen Verteilungen}
Ist der unbekannte Parameter $\Theta$ selbst stetig verteilt, dann arbeiten wir mit Dichten statt mit einzelnen Wahrscheinlichkeiten.
Für Prior-Dichte $p(\theta)$ und Likelihood-Dichte $f(x \mid \theta)$ gilt nach Beobachtung $X=x$:
\begin{align*}
p(\theta \mid x)
= \frac{f(x \mid \theta)\, p(\theta)}{\int_{-\infty}^{\infty} f(x \mid u)\, p(u)\, du}.
\end{align*}
Bei stetigen Verteilungen ist nämlich $P(X=x)=0$; relevant ist daher die Dichte an der beobachteten Stelle $x$.
\end{block}

 \end{frame}


\begin{frame}[shrink=5]
    \frametitle{Diskrete Wahrscheinlichkeitsverteilung}
\framesubtitle{}

\begin{block}{Beispiel: Raumtemperatur und Thermometer}
Sei $\Theta$ die wahre Raumtemperatur in Grad Celsius. Vor der Messung glauben wir, dass die Temperatur meistens in der Nähe von $20^\circ$C liegt:
\begin{align*}
\Theta \sim \mathcal N(20,2^2).
\end{align*}
Das Thermometer misst nicht perfekt, sondern mit einem Fehler von etwa $1^\circ$C. Also modellieren wir
\begin{align*}
X \mid \Theta=\theta \sim \mathcal N(\theta,1^2).
\end{align*}
Wenn das Thermometer nun $X=24$ misst, dann ergibt sich als Posterior wieder eine Normalverteilung:
\begin{align*}
\Theta \mid X=24 \sim \mathcal N(23.2,0.8).
\end{align*}
\end{block}

\begin{block}{Interpretation}
Der Prior sagt: Vor der Messung erwarten wir ungefähr $20^\circ$C.
Die Likelihood sagt: Ein Messwert von $24^\circ$C ist besonders plausibel, wenn die wahre Temperatur in der Nähe von $24^\circ$C liegt.
Der Posterior kombiniert beides: Nach der Messung halten wir Werte um $23.2^\circ$C für am plausibelsten.
\end{block}

 \end{frame}



\begin{frame}
    \frametitle{Diskrete Wahrscheinlichkeitsverteilung}
\framesubtitle{}

\begin{block}{Erinnerung Stochastische Unabhängigkeit}
Zwei Ereignisse $A,B$ heißen stochastisch unabhängig, falls
\begin{align*}
P(A \cap B) = P(A) \cdot P(B)
\end{align*}
gilt.  Gleichbedeutend damit ist  $P(A | B) = P(A)$ und $P(B  | A) = P(B)$.
\end{block}

 \end{frame}

(Typische Wörter wären zum Beispiel "reich",  "casino", "Vergrösserung" ...)
\end{block}

\begin{frame}
    \frametitle{Diskrete Wahrscheinlichkeitsverteilung}
\framesubtitle{}

\begin{block}{Naiver Bayes'scher Spam Filter}

Aus einer Datenbank kann man das Vorkommen dieser Wörter in Spam und Ham Mails zählen und damit empirisch die Wahrscheinlichkeiten $P(A_i | S)$ und $P(A_i | H) $ des Vorkommens dieser Wörter in Spam und Ham Mails ermitteln.  Wir gehen davon aus, dass es sich bei der Mail  prinzipiell mit  Wahrscheinlichkeit $P(E= S) = P(E= H)= \frac{1}{2}$  um eine erwünschte  Mail $H$ oder eine unerwünschte Mail $S$  handeln kann. 
\end{block}

 \end{frame}



\begin{frame}
    \frametitle{Diskrete Wahrscheinlichkeitsverteilung}
\framesubtitle{}

\begin{block}{Naiver Bayes'scher Spam Filter}
 Wir machen zudem die (naive) Annahme, dass das Vorkommen der Wörter  stochastisch unabhängig ist, also 
\begin{align*}
P(A_1 \cap \cdots \cap A_n | S) = P(A_1 | S) \cdot P(A_2 | S) \cdots P(A_n | S) \\
P(A_1 \cap \cdots \cap A_n | H) = P(A_1 | H) \cdot P(A_2 | H) \cdots P(A_n | H)
\end{align*}
gilt.
\end{block}

 \end{frame}


\begin{frame}
    \frametitle{Diskrete Wahrscheinlichkeitsverteilung}
\framesubtitle{}

\begin{block}{Naiver Bayes'scher Spam Filter}
Mit der Formel von Bayes, der totalen Wahrscheinlichkeit  und kürzen, da $P(S) = P(H)$ können wir somit berechnen
\begin{align*}
& P(E=S |  A_1 \cap \cdots \cap A_n) = \frac{P(A_1 \cap \cdots \cap A_n | S) \cdot P(S)}{P(A_1 \cap \cdots \cap A_n)} \\
&=  \frac{P(A_1 | S) \cdots P(A_n | S) }{P(A_1 \cap \cdots \cap A_n | H) + P(A_1 \cap \cdots \cap A_n | S)} \\
&=  \frac{P(A_1 | S) \cdots P(A_n | S) }{P(A_1 | H) \cdots P(A_n | H)  + P(A_1 | S) \cdots P(A_n | S) } \\
\end{align*}
Bemerkung: $P(E=H |  A_1 \cap \cdots \cap A_n) = 1- P(E=S |  A_1 \cap \cdots \cap A_n) $
\end{block}



 \end{frame}


\begin{frame}[shrink=5]
    \frametitle{Diskrete Wahrscheinlichkeitsverteilung}
\framesubtitle{}

\begin{block}{Spamfilter als Bayes'sche Inferenz}
Der naive Bayes-Filter ist ein klassisches Beispiel für Bayes'sche Inferenz:
\begin{align*}
&\text{Hypothesen: } H=\text{Ham},\ S=\text{Spam}, \\
&\text{Daten: } D = A_1 \cap \cdots \cap A_n.
\end{align*}
Aus Prior und Likelihoods berechnen wir die Posterior-Wahrscheinlichkeiten
\begin{align*}
P(S \mid D), \qquad P(H \mid D).
\end{align*}
\end{block}

\begin{block}{Einordnung}
\begin{itemize}
\item Prior: $P(S), P(H)$ beschreibt, wie wahrscheinlich Spam oder Ham vor den Wörtern ist.
\item Likelihood: $P(D \mid S), P(D \mid H)$ misst, wie gut die beobachteten Wörter zu Spam oder Ham passen.
\item Posterior: $P(S \mid D), P(H \mid D)$ kombiniert beides nach der Bayes-Formel.
\item Entscheidung: Klassifiziere als Spam, falls $P(S \mid D) > P(H \mid D)$.
\end{itemize}
\end{block}

 \end{frame}

    \frametitle{Erwartungswert}
\framesubtitle{}
\begin{block}{Eigenschaften}
Sind $X,Y : \Omega \to \mathbb{R}^n$   reelle, integrierbare  Zufallsvariablen und $a,b \in \mathbb{R}$ konstant, so gilt:
\begin{align*}
& \mathbb{E}(a \cdot X + b \cdot Y) = a \cdot \mathbb{E}(X) + b \cdot \mathbb{E}(Y) \\
& X(x) \leq Y(x) \;  \forall x \in \Omega \Rightarrow \mathbb{E}(X) \leq \mathbb{E}(Y) \\
& X ,Y \text{ stoch. unabhängig} \Rightarrow   \mathbb{E}(X \cdot Y) =  \mathbb{E}(X) \cdot  \mathbb{E}(Y) \\
& \mathbb{E} (1_A) = P (A)
\end{align*}
\end{block}
 \end{frame}

 \begin{frame}
    \frametitle{Erwartungswert}
\framesubtitle{}
\begin{block}{Verschiebungssatz (Erwartungswert)}
Sei $X : \Omega \to \mathbb{R}$ eine reelle Zufallsvariable und $g : \mathbb{R} \to \mathbb{R}$ messbar. Dann gilt:
\begin{align*}
\mathbb{E}(g(X)) = \int_{\mathbb{R}} g(x) \, dP_X(x)
\end{align*}
wobei $P_X$ die Verteilung von $X$ ist. Für eine Dichte $f_X$:
\begin{align*}
\mathbb{E}(g(X)) = \int_{\mathbb{R}} g(x) \, f_X(x) \, dx
\end{align*}
Man muss also \textbf{nicht} die Verteilung von $g(X)$ bestimmen, sondern kann direkt mit der Verteilung von $X$ rechnen.
\end{block}
 \end{frame}

 \begin{frame}
    \frametitle{Erwartungswert}
\framesubtitle{}
\begin{block}{Beweis des Verschiebungssatzes (Schritt 1: Indikatorfunktionen)}
Für $g = 1_B$ mit einer Borel-Menge $B \subseteq \mathbb{R}$:
\begin{align*}
\mathbb{E}(1_B(X)) = P(X \in B) = P_X(B) = \int_{\mathbb{R}} 1_B(x) \, dP_X(x) \; \checkmark
\end{align*}
Dies folgt direkt aus der Definition des Bildmaßes $P_X(B) = P(X^{-1}(B))$.
\end{block}

\begin{block}{Beweis (Schritt 2: Einfache Funktionen)}
Für $g = \sum_{i=1}^n a_i \cdot 1_{B_i}$ folgt aus Schritt 1 und der Linearität des Integrals:
\begin{align*}
\mathbb{E}(g(X)) = \sum_{i=1}^n a_i \, P_X(B_i) = \int_{\mathbb{R}} g(x) \, dP_X(x)
\end{align*}
\end{block}
 \end{frame}

 \begin{frame}
    \frametitle{Erwartungswert}
\framesubtitle{}
\begin{block}{Beweis (Schritt 3: Allgemeine messbare $g \geq 0$)}
Es existieren einfache Funktionen $g_n \nearrow g$ (monoton wachsend). Mit dem Satz der monotonen Konvergenz:
\begin{align*}
\mathbb{E}(g(X)) = \lim_n \mathbb{E}(g_n(X)) \overset{\text{Schritt 2}}{=} \lim_n \int g_n \, dP_X = \int g \, dP_X
\end{align*}
\end{block}

\begin{block}{Beweis (Schritt 4: Allgemeines $g$)}
Zerlege $g = g^+ - g^-$ mit $g^+ = \max(g,0)$ und $g^- = \max(-g,0)$.
Wende Schritt 3 auf $g^+$ und $g^-$ an und nutze die Linearität. $\qed$
\end{block}
 \end{frame}

 \begin{frame}
    \frametitle{Erwartungswert}
\framesubtitle{}
\begin{block}{Varianz}
Für eine reelle Zufallsvariable ist die Varianz definiert durch
$$ \mathbb{V} (X) :=  \mathbb{E}( (X - \mathbb{E}(X))^2) \; .$$
\end{block}
\begin{block}{Verschiebungssatz}

\begin{align*}
 \mathbb{V}(X) & = \mathbb{E}(X^2 - 2X \mathbb{E}(X) + \mathbb{E}(X)^2) = \mathbb{E}(X^2) - 2 \mathbb{E}(X)^2 +  \mathbb{E}(X)^2 \\
& =  \mathbb{E}(X^2) -  \mathbb{E}(X)^2 \\
\end{align*}
\end{block}
 \end{frame}

 \begin{frame}
    \frametitle{Varianz}
\framesubtitle{}
\begin{block}{Eigenschaften der Varianz}
Sind $X,Y : \Omega \to \mathbb{R}$ reelle, quadratintegrierbare Zufallsvariablen und $a,b \in \mathbb{R}$ konstant, so gilt:
\begin{align*}
& \mathbb{V}(a \cdot X + b) = a^2 \cdot \mathbb{V}(X) \\
& \mathbb{V}(X) \geq 0, \quad \mathbb{V}(X) = 0 \Leftrightarrow X = \mathbb{E}(X) \text{ f.s.} \\
& X, Y \text{ stoch. unabhängig} \Rightarrow \mathbb{V}(X + Y) = \mathbb{V}(X) + \mathbb{V}(Y) \\
& \mathbb{V}(X + Y) = \mathbb{V}(X) + \mathbb{V}(Y) + 2\,\mathcal{C}(X,Y) \quad \text{(allgemein)}
\end{align*}
\end{block}
 \end{frame}

 \begin{frame}
    \frametitle{Erwartungswert}
\framesubtitle{}
\begin{block}{Kovarianz}
Für  reelle Zufallsvariable $X,Y$ ist die Kovarianz definiert durch
$$ \mathcal{C} (X,Y) :=  \mathbb{E}( (X - \mathbb{E}(X)) (Y - \mathbb{E}(Y))) \; .$$
\end{block}

\begin{block}{Kovarianz}
    Per Definition ist
    $$ \mathcal{C} (X,X) :=  \mathbb{V}(X).$$
    \end{block}
 \end{frame}

\begin{frame}
    \frametitle{Erwartungswert}
\framesubtitle{}
\begin{block}{Markov Ungleichung}
Sei $Y : \Omega \to \mathbb{R}$  eine  reelle, integrierbare  Zufallsvariable und $f : [0, \infty) \to [0, \infty)$ monoton wachsend.
Dann gilt für alle $\epsilon > 0$ mit $f(\epsilon) > 0$
\begin{align*}
P (|Y |  \geq \epsilon) \leq \frac{\mathbb{E} (f \circ |Y|)}{f(\epsilon)}
\end{align*}
\end{block}
\begin{block}{Beweis}
Da $f(\epsilon) 1_{\{ |Y| \geq  \epsilon \} } \leq f \circ |Y|$ folgt
\begin{align*}
f(\epsilon) P(|Y| \geq \epsilon) = & f(\epsilon) \mathbb{E}(1_{\{ |Y| \geq  \epsilon \} }) = \mathbb{E}( f(\epsilon) 1_{\{ |Y| \geq  \epsilon \} }) \\
\leq & \mathbb{E}( f \circ |Y|)
\end{align*}
\end{block}
 \end{frame}






\begin{frame}
    \frametitle{Erwartungswert}
\framesubtitle{}
\begin{block}{Tschebyscheff-Ungleichung}
Für eine reelle, integrierbare und quadratintegrierbare  Zufallsvariable $Y : \Omega \to \mathbb{R}$  gilt:
\begin{align*}
P (|Y  - \mathbb{E} (Y)|  \geq \epsilon) \leq \frac{\mathbb{V} (Y)}{ \epsilon^2} 
\end{align*}
\end{block}
\begin{block}{Beweis}
Folgt direkt aus der Markov-Ungleichung mit $Y' = Y -\mathbb{E}(Y)$ und $f(x) = x^2$
\end{block}
 \end{frame}



\begin{frame}
    \frametitle{Erwartungswert}
\framesubtitle{}
\begin{block}{Schwaches Gesetz der großen Zahlen}
Seien $X_i : \Omega \to \mathbb{R}$ unabhängige, reelle Zufallsvariablen (uiv, iid(englisch)) mit $\mathbb{E}(X_i) = \mu < \infty$ und $\mathbb{V}(X_i) = \sigma < \infty$, dann gilt
\begin{align*}
P \bigl  ( \bigl | \frac{1}{n} \sum_{i=1}^{n} X_i - \mu \bigr |  \geq \epsilon \bigr) \leq \frac{\sigma}{ n \cdot \epsilon^2} \; \; \underset{n \to \infty}{\longrightarrow} 0
\end{align*}
(stochastische Konvergenz). 
\end{block}
 \end{frame}


\begin{frame}
    \frametitle{Erwartungswert}
\framesubtitle{}
\begin{block}{Beweis}
Mit $Y_n =  \frac{1}{n} \sum_{i=1}^{n}  X_i - \mu$ ist $\mathbb{E}(Y_n) =  \frac{1}{n} \sum_{i=1}^{n} \mathbb{E}( X_i - \mu) = 0$ und 
$\mathbb{V}(Y_n) =  \frac{1}{n^2} \sum_{i=1}^{n} \mathbb{V}( X_i ) = \frac{\sigma}{n}$. Aus der Tschebyscheff-Ungleichung folgt die Behauptung.
\end{block}
 \end{frame}

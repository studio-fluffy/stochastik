\begin{frame}
    \frametitle{Markov-Ketten}
\framesubtitle{Motivation}
    \begin{block}{Idee}
Viele Prozesse in der Informatik und im Alltag können als Folge von Zuständen modelliert werden, bei denen der nächste Zustand nur vom aktuellen Zustand abhängt -- nicht von der gesamten Vergangenheit.
\end{block}
\begin{block}{Beispiele}
\begin{itemize}
\item Websurfen: Die nächste Seite hängt nur von der aktuellen ab (PageRank).
\item Wetter: Morgen Regen/Sonne hängt nur vom heutigen Wetter ab.
\item Kartenmischen: Die nächste Anordnung hängt nur von der aktuellen ab.
\item Warteschlangen: Anzahl Kunden hängt nur vom aktuellen Zustand ab.
\end{itemize}
\end{block}
 \end{frame}


%% =============================================
%% DEFINITION STOCHASTISCHE MATRIX
%% =============================================

\begin{frame}
    \frametitle{Markov-Ketten}
\framesubtitle{Stochastische Matrix}
    \begin{block}{Definition (Stochastische Matrix)}
Eine Matrix $P = (p_{ij})_{i,j \in S}$ mit Zustandsraum $S$ heißt \textbf{stochastische Matrix}, falls:
\begin{enumerate}
\item $p_{ij} \geq 0$ für alle $i, j \in S$
\item $\sum_{j \in S} p_{ij} = 1$ für alle $i \in S$ \quad (Zeilensumme $= 1$)
\end{enumerate}
\end{block}
\begin{block}{Interpretation}
$p_{ij}$ ist die Wahrscheinlichkeit, vom Zustand $i$ in den Zustand $j$ zu wechseln.
\end{block}
 \end{frame}

%% =============================================
%% DEFINITION MARKOV-KETTE
%% =============================================

\begin{frame}
    \frametitle{Markov-Ketten}
\framesubtitle{Definition}
    \begin{block}{Definition (Markov-Kette)}
Eine Folge $(X_n)_{n \in \N_0}$ von Zufallsvariablen mit Werten in $S$ heißt \textbf{Markov-Kette} mit Übergangsmatrix $P$, falls:
\[
\PP(X_{n+1} = j \mid X_n = i, X_{n-1}, \ldots, X_0) = \PP(X_{n+1} = j \mid X_n = i) = p_{ij}
\]
für alle $n \in \N_0$ und alle $i, j \in S$.
\end{block}
\begin{block}{Markov-Eigenschaft (Gedächtnislosigkeit)}
Die Zukunft hängt nur von der Gegenwart ab, nicht von der Vergangenheit!
\end{block}
 \end{frame}

%% =============================================
%% n-SCHRITT ÜBERGANGSWAHRSCHEINLICHKEITEN
%% =============================================

\begin{frame}
    \frametitle{Markov-Ketten}
\framesubtitle{$n$-Schritt Übergangswahrscheinlichkeiten}
    \begin{block}{Satz (Chapman-Kolmogorov)}
Die Wahrscheinlichkeit, in $n$ Schritten von $i$ nach $j$ zu gelangen:
\[
p_{ij}^{(n)} = \PP(X_n = j \mid X_0 = i) = (P^n)_{ij}
\]
Also: Einfach die Matrix $n$-mal mit sich selbst multiplizieren!
\end{block}
\begin{block}{Beispiel (Wetter)}
Wie ist das Wetter übermorgen, wenn heute Sonne?
\[
P^2 = \begin{pmatrix} 0.8 & 0.2 \\ 0.4 & 0.6 \end{pmatrix}^2 = \begin{pmatrix} 0.72 & 0.28 \\ 0.56 & 0.44 \end{pmatrix}
\]
$\Rightarrow$ Wahrscheinlichkeit für Sonne übermorgen: $0.72$.
\end{block}
 \end{frame}


%% =============================================
%% BEWEIS CHAPMAN-KOLMOGOROV
%% =============================================

\begin{frame}
    \frametitle{Markov-Ketten}
\framesubtitle{Beweis: Chapman-Kolmogorov}
    \begin{block}{Beweis ($P^{m+n} = P^m \cdot P^n$)}
\small
\begin{align*}
p_{ij}^{(m+n)} &= \PP(X_{m+n} = j \mid X_0 = i) & &\text{(Definition)}\\
&= \sum_{k \in S} \PP(X_{m+n} = j, X_m = k \mid X_0 = i) & &\text{(Tot.\ W'keit)}\\
&= \sum_{k \in S} \PP(X_{m+n}\! =\! j \mid X_m\! =\! k, X_0\! =\! i) \cdot p_{ik}^{(m)} & &\text{(Def.\ bed.\ W'keit)}\\
&= \sum_{k \in S} \PP(X_{m+n} = j \mid X_m = k) \cdot p_{ik}^{(m)} & &\text{(Markov-Eig.)}\\
&= \sum_{k \in S} p_{ik}^{(m)} \cdot p_{kj}^{(n)} & &\text{(Homogenität)} \quad \square
\end{align*}
\end{block}
 \end{frame}


%% =============================================
%% NOTATIONS-IDENTIFIKATION P^{(m)} = P^m
%% =============================================

\begin{frame}
    \frametitle{Markov-Ketten}
\framesubtitle{Bemerkung: $P^{(m)} = P^m$}
\begin{block}{Was haben wir gerade gezeigt?}
Mit $P^{(m)} := \bigl(p_{ij}^{(m)}\bigr)_{i,j \in S}$ liefert die Rechnung komponentenweise
\[
P^{(m+n)} = P^{(m)} \cdot P^{(n)} \qquad (\star)
\]
-- eine Aussage über die $P^{(\cdot)}$, noch nicht über Matrix\emph{potenzen} $P^m$!
\end{block}
\begin{block}{Induktion: $P^{(m)} = P^m$}
\textbf{Basis} ($m=1$): $P^{(1)}_{ij} = p_{ij}$, also $P^{(1)} = P$.\\
\textbf{Schritt}: Mit $n=1$ in $(\star)$ folgt
$P^{(m+1)} = P^{(m)} \cdot P \stackrel{\text{IV}}{=} P^m \cdot P = P^{m+1}.$
\end{block}
\begin{block}{Folgerung}
$P^{m+n} = P^{(m+n)} \stackrel{(\star)}{=} P^{(m)} \cdot P^{(n)} = P^m \cdot P^n. \qquad \square$
\end{block}
 \end{frame}


%% =============================================
%% VERTEILUNG ZUM ZEITPUNKT n
%% =============================================

\begin{frame}
    \frametitle{Markov-Ketten}
\framesubtitle{Einschub: Linksmultiplikation Zeilenvektor $\cdot$ Matrix}
\begin{block}{Definition}
Für einen \textbf{Zeilenvektor} $\mu = (\mu_1, \ldots, \mu_N) \in \R^{1 \times N}$ und $P = (p_{ij}) \in \R^{N \times N}$ ist $\mu \cdot P \in \R^{1 \times N}$ definiert durch
\[
(\mu \cdot P)_j \;=\; \sum_{i=1}^{N} \mu_i \, p_{ij}, \qquad j = 1, \ldots, N.
\]
Das ist das gewöhnliche Matrixprodukt einer $1 \!\times\! N$- mit einer $N \!\times\! N$-Matrix.
\end{block}
\begin{block}{Wahrscheinlichkeitsinterpretation (totale W'keit)}
Mit $\mu_i = \PP(X_n = i)$ und $p_{ij} = \PP(X_{n+1} = j \mid X_n = i)$:
\[
(\mu P)_j \;=\; \sum_i \PP(X_n = i)\, \PP(X_{n+1} = j \mid X_n = i) \;=\; \PP(X_{n+1} = j).
\]
\end{block}
 \end{frame}


\begin{frame}
    \frametitle{Markov-Ketten}
\framesubtitle{Verteilung zum Zeitpunkt $n$}
    \begin{block}{Anfangsverteilung}
Die Verteilung von $X_0$ wird als \textbf{Anfangsverteilung} $\mu_0$ bezeichnet:
\[
\mu_0(i) = \PP(X_0 = i), \quad i \in S
\]
Als Zeilenvektor: $\mu_0 = (\mu_0(1), \mu_0(2), \ldots)$
\end{block}
\begin{block}{Verteilung nach $n$ Schritten}
\[
\mu_n = \mu_0 \cdot P^n
\]
\end{block}
\begin{block}{Beispiel (Wetter)}
Start: $\mu_0 = (1, 0)$ (heute sicher Sonne). Nach 2 Tagen:
\[
\mu_2 = (1, 0) \cdot P^2 = (0.72, \; 0.28)
\]
\end{block}
 \end{frame}


%% =============================================
%% INVARIANTE VERTEILUNG - DEFINITION
%% =============================================

\begin{frame}
    \frametitle{Markov-Ketten}
\framesubtitle{Invariante (stationäre) Verteilung}
    \begin{block}{Definition (Invariante Verteilung)}
Ein Wahrscheinlichkeitsvektor $\pi = (\pi_1, \pi_2, \ldots)$ heißt \textbf{invariante} (oder \textbf{stationäre}) \textbf{Verteilung}, falls:
\[
\pi \cdot P = \pi
\]
d.h. $\pi$ ist ein Linkseigenvektor von $P$ zum Eigenwert $1$.
\end{block}
\begin{block}{Interpretation}
Startet die Kette mit Verteilung $\pi$, bleibt sie für alle Zeiten bei $\pi$:
\[
\mu_0 = \pi \;\Rightarrow\; \mu_n = \pi \cdot P^n = \pi \quad \text{für alle } n.
\]
Das System befindet sich im Gleichgewicht.
\end{block}
 \end{frame}


%% =============================================
%% BEISPIEL: INVARIANTE VERTEILUNG BERECHNEN
%% =============================================

\begin{frame}
    \frametitle{Markov-Ketten}
\framesubtitle{Beispiel: Invariante Verteilung berechnen}
    \begin{block}{Wetter-Beispiel}
Gesucht: $\pi = (\pi_S, \pi_R)$ mit $\pi P = \pi$ und $\pi_S + \pi_R = 1$.
\end{block}
\begin{align*}
\pi P = \pi &\Leftrightarrow \begin{cases} 0.8 \pi_S + 0.4 \pi_R = \pi_S \\ 0.2 \pi_S + 0.6 \pi_R = \pi_R \end{cases} \\
&\Leftrightarrow -0.2 \pi_S + 0.4 \pi_R = 0 \\
&\Leftrightarrow \pi_S = 2 \pi_R
\end{align*}
Mit $\pi_S + \pi_R = 1$: \quad $2\pi_R + \pi_R = 1$ \quad $\Rightarrow$ \quad $\pi_R = \frac{1}{3}$, \; $\pi_S = \frac{2}{3}$.

\[
\boxed{\pi = \left(\frac{2}{3}, \frac{1}{3}\right)}
\]

\textbf{Interpretation:} Langfristig ist es $2/3$ der Zeit sonnig und $1/3$ regnerisch.
 \end{frame}

\begin{frame}
    \frametitle{Markov-Ketten}
\framesubtitle{Irreduzibilität}
    \begin{block}{Definition (Irreduzibel)}
Eine Markov-Kette heißt \textbf{irreduzibel}, falls man von jedem Zustand $i$ jeden anderen Zustand $j$ erreichen kann:
\[
\forall\, i,j \in S \;\exists\, n \geq 1: \quad p_{ij}^{(n)} > 0
\]
\end{block}

\end{columns}

 \end{frame}

\begin{frame}
    \frametitle{Markov-Ketten}
\framesubtitle{Aperiodizität}
    \begin{block}{Definition (Periode)}
Die \textbf{Periode} eines Zustands $i$ ist:
\[
d(i) = \mathrm{ggT}\{n \geq 1 : p_{ii}^{(n)} > 0\}
\]
Die Kette heißt \textbf{aperiodisch}, falls $d(i) = 1$ für alle $i \in S$.
\end{block}
\begin{columns}
\begin{column}{0.5\textwidth}
\begin{center}
\textbf{Aperiodisch} ($d=1$):\\[0.3cm]

    \node[state] (1) {$1$};
    \node[state] (2) [right of=1] {$2$};
    \path (1) edge[loop left] node{} (1)
          (1) edge[bend left] node{} (2)
          (2) edge[bend left] node{} (1);
\end{tikzpicture}
\end{center}
\end{column}
\begin{column}{0.5\textwidth}
\begin{center}
\textbf{Periodisch} ($d=2$):\\[0.3cm]
\begin{tikzpicture}[->, >=stealth, auto, semithick, node distance=2.5cm, scale=0.8]
    \node[state] (1) {$1$};
    \node[state] (2) [right of=1] {$2$};
    \path (1) edge[bend left] node{} (2)
          (2) edge[bend left] node{} (1);
\end{tikzpicture}
\end{center}
\end{column}
\end{columns}
\vspace{0.3cm}
\textbf{Hinweis:} Eine Selbstschleife ($p_{ii} > 0$) garantiert $d(i)=1$, ist aber nicht notwendig.
 \end{frame}

%% BEISPIEL PERIODISCH

\begin{frame}
    \frametitle{Markov-Ketten}
\framesubtitle{Beispiel: Periodische Kette}
    \begin{block}{Beispiel mit 2 Zuständen}
\[
P = \begin{pmatrix} 0 & 1 \\ 1 & 0 \end{pmatrix}
\]
\end{block}
Die Kette springt \emph{immer} zwischen den Zuständen hin und her:
\[
1 \to 2 \to 1 \to 2 \to \ldots
\]
\begin{itemize}
\item Rückkehr zu Zustand 1 nur nach $2, 4, 6, \ldots$ Schritten.
\item $d(1) = \mathrm{ggT}\{2, 4, 6, \ldots\} = 2$. \quad $\Rightarrow$ Periodisch!
\item Invariante Verteilung: $\pi = (1/2, 1/2)$, aber $P^n$ konvergiert \textbf{nicht}:
\[
P^{2k} = \begin{pmatrix} 1 & 0 \\ 0 & 1 \end{pmatrix}, \qquad
P^{2k+1} = \begin{pmatrix} 0 & 1 \\ 1 & 0 \end{pmatrix}
\]
(oszilliert zwischen zwei Werten).
\end{itemize}
 \end{frame}


%% =============================================
%% HAUPTSATZ: EXISTENZ UND EINDEUTIGKEIT
%% =============================================

\begin{frame}
    \frametitle{Markov-Ketten}
\framesubtitle{Hauptsatz über invariante Verteilungen}
    \begin{block}{Satz (Existenz und Eindeutigkeit)}
Sei $(X_n)$ eine \textbf{irreduzible} Markov-Kette mit endlichem Zustandsraum $S$. Dann:
\begin{enumerate}
\item Es existiert \textbf{genau eine} invariante Verteilung $\pi$.
\item $\pi_i > 0$ für alle $i \in S$.
\item Ist die Kette zusätzlich \textbf{aperiodisch}, so gilt:
\[
\lim_{n \to \infty} \mu_0 P^n = \pi
\]
d.h.\ Konvergenz gegen $\pi$, unabhängig vom Startzustand.
\end{enumerate}
\end{block}
 \end{frame}


%% =============================================
%% BEWEIS TEIL 1: EXISTENZ
%% =============================================

\begin{frame}
    \frametitle{Markov-Ketten}
\framesubtitle{Beweis -- Teil 1: Existenz}
    \begin{block}{Cesàro-Mittel}
$\overline{P}_n := \frac{1}{n} \sum_{k=0}^{n-1} P^k$ \quad (Konvexkombination stochastischer Matrizen $\Rightarrow$ stochastisch)
\end{block}
\begin{itemize}
\item Jede Zeile von $\overline{P}_n$ liegt im \textbf{Simplex} $\Delta = \{x \in \R^{|S|} : x_i \geq 0,\; \sum x_i = 1\} \subset [0,1]^{|S|}$.
\item $|S| < \infty$ $\Rightarrow$ $\Delta$ beschränkt + abgeschlossen $\Rightarrow$ \textbf{kompakt} (Heine-Borel).
\item \textbf{Bolzano-Weierstraß}: $\exists$ konvergente Teilfolge $\overline{P}_{n_k} \to Q$.
\item Jede Zeile $q_i$ von $Q$ ist invariant, denn:
\[
\overline{P}_{n_k} \cdot P = \overline{P}_{n_k} + \tfrac{1}{n_k}(P^{n_k} - I) \;\xrightarrow{k\to\infty}\; Q + 0 = Q
\]
Also $Q \cdot P = Q$, d.h. $q_i \cdot P = q_i$. \hfill $\square$
\end{itemize}
 \end{frame}


%% =============================================
%% BEWEIS TEIL 2: POSITIVITÄT
%% =============================================

\begin{frame}
    \frametitle{Markov-Ketten}
\framesubtitle{Beweis -- Teil 2: $\pi_i > 0$ für alle $i$}
    \begin{block}{Alle Einträge von $\pi$ sind positiv}
Sei $\pi$ eine invariante Verteilung und $i \in S$ beliebig.
\end{block}
\begin{itemize}
\item Da die Kette irreduzibel ist, gibt es $j$ mit $\pi_j > 0$ und $m \geq 1$ mit $p_{ji}^{(m)} > 0$.
\item Aus $\pi = \pi P^m$ folgt:
\[
\pi_i = \sum_{k \in S} \pi_k \cdot p_{ki}^{(m)} \geq \pi_j \cdot p_{ji}^{(m)} > 0
\]
\item Da mindestens ein $\pi_j > 0$ existiert ($\pi$ ist W-Vektor), folgt $\pi_i > 0$ für alle $i$.
\end{itemize}
 \end{frame}


%% =============================================
%% BEWEIS TEIL 3: EINDEUTIGKEIT
%% =============================================

\begin{frame}
    \frametitle{Markov-Ketten}
\framesubtitle{Beweis -- Teil 3: Eindeutigkeit}
    \begin{block}{Die invariante Verteilung ist eindeutig}
Angenommen, $\pi$ und $\tilde{\pi}$ sind zwei invariante Verteilungen.
\end{block}
Definiere $f := \pi - \tilde{\pi}$. Dann gilt:
\begin{itemize}
\item $f \cdot P = f$ \quad (Differenz zweier Fixpunkte ist Fixpunkt)
\item $\sum_i f_i = 1 - 1 = 0$
\end{itemize}
Sei $i^* = \arg\max_i f_i$. Aus $f = fP$ folgt:
\[
f_{i^*} = \sum_j p_{i^* j} f_j \leq f_{i^*} \cdot \underbrace{\sum_j p_{i^* j}}_{= 1} = f_{i^*}
\]
Gleichheit nur wenn $f_j = f_{i^*}$ für alle $j$ mit $p_{i^* j} > 0$.
 \end{frame}


\begin{frame}
    \frametitle{Markov-Ketten}
\framesubtitle{Beweis -- Teil 3: Eindeutigkeit (Fortsetzung)}
    \begin{block}{Schluss per Irreduzibilität}
Wir haben: $f_j = f_{i^*}$ für alle $j$ mit $p_{i^*j} > 0$.
\end{block}
\begin{itemize}
\item Wenn $f_j = f_{i^*}$ und $p_{jk} > 0$, dann auch $f_k = f_{i^*}$.
\item Da die Kette \textbf{irreduzibel} ist, erreicht man von $i^*$ jeden Zustand.
\item Also: $f_k = f_{i^*}$ für \textbf{alle} $k \in S$.
\item Da $\sum_k f_k = 0$ und alle $f_k$ gleich: $|S| \cdot f_{i^*} = 0 \Rightarrow f_{i^*} = 0$.
\item Damit $f = 0$, d.h. $\pi = \tilde{\pi}$. \hfill $\square$
\end{itemize}
 \end{frame}


%% =============================================
%% BEWEIS TEIL 4: KONVERGENZ
%% =============================================

\begin{frame}
    \frametitle{Markov-Ketten}
\framesubtitle{Beweis -- Teil 4: Konvergenz (Skizze)}
    \begin{block}{Bei Aperiodizität: $P^n \to \Pi$ (alle Zeilen $= \pi$)}
\end{block}
\begin{itemize}
\item \textbf{Variationsabstand}: $\|u - v\|_{TV} = \frac{1}{2}\sum_i |u_i - v_i|$.
\item Für die Verteilung $e_i P^n$ (Start in $i$):
\[
\|e_i P^n - \pi\|_{TV} \leq \|e_i - \pi\|_{TV} \cdot \rho^n
\]
wobei $\rho < 1$ (Kontraktion bei Irreduzibilität + Aperiodizität).
\item Da $\rho^n \to 0$: Konvergenz gegen $\pi$ für jeden Startzustand. \hfill $\square$
\end{itemize}
 \end{frame}


\begin{frame}
    \frametitle{Markov-Ketten}
\framesubtitle{Warum gibt es ein $\rho < 1$?}
    \begin{block}{Schlüsselidee}
Endlich + irreduzibel + aperiodisch $\Rightarrow$ es gibt ein $m \geq 1$ mit $P^m > 0$.\\
Also hat nach $m$ Schritten jede Zeile von $P^m$ überall etwas Masse.
\end{block}
\begin{block}{Folge}
Dann gibt es ein $c > 0$ mit
\[
\|\mu P^m - \nu P^m\|_{TV} \leq (1-c)\,\|\mu-\nu\|_{TV}
\]
für alle Verteilungen $\mu,\nu$. Durch Wiederholung folgt
\[
\|\mu P^n - \pi\|_{TV} \leq C\rho^n \qquad (0<\rho<1)
\]
also geometrische Konvergenz.
\end{block}
 \end{frame}

%% =============================================
%% DETAILED BALANCE (REVERSIBILITÄT)
%% =============================================

\begin{frame}
    \frametitle{Markov-Ketten}
\framesubtitle{Detailed Balance -- Reversibilität}
    \begin{block}{Definition (Detailed Balance)}
$\pi$ erfüllt die \textbf{Detailed-Balance-Bedingung} bzgl. $P$, falls:
\[
\pi_i \cdot p_{ij} = \pi_j \cdot p_{ji} \quad \text{für alle } i, j \in S
\]
\end{block}
\begin{block}{Satz: Detailed Balance $\Rightarrow$ $\pi$ invariant}
\[
(\pi P)_j = \sum_i \pi_i p_{ij} \overset{\text{DB}}{=} \sum_i \pi_j p_{ji} = \pi_j \underbrace{\sum_i p_{ji}}_{=1} = \pi_j \quad \square
\]
\end{block}
\begin{block}{Intuition: Flussgleichgewicht}
$\pi_i p_{ij}$ = ``Fluss'' von $i$ nach $j$ im Gleichgewicht.\\
DB: Fluss $i \to j$ = Rückfluss $j \to i$ für \emph{jedes} Paar $\Rightarrow$ Gleichgewicht.
\end{block}
 \end{frame}

%% DETAILED BALANCE BEISPIEL

\begin{frame}

\framesubtitle{Detailed Balance -- Beispiel}
    \begin{block}{Wetter-Beispiel: Prüfe Detailed Balance}
$\pi = (2/3, 1/3)$, \quad $P = \begin{pmatrix} 0.8 & 0.2 \\ 0.4 & 0.6\end{pmatrix}$
\end{block}
Prüfe $\pi_S \cdot p_{SR} = \pi_R \cdot p_{RS}$:
\[
\frac{2}{3} \cdot 0.2 = \frac{2}{15} \qquad \text{und} \qquad \frac{1}{3} \cdot 0.4 = \frac{2}{15} \quad \checkmark
\]
$\Rightarrow$ Das Wetter-Modell ist \textbf{reversibel}.

\vspace{0.2cm}
\textbf{Nicht jede Kette ist reversibel!} Kreislauf $1 \to 2 \to 3 \to 1$:
\[
P = \begin{pmatrix} 0 & 1 & 0 \\ 0 & 0 & 1 \\ 1 & 0 & 0 \end{pmatrix}, \quad
\pi = (1/3, 1/3, 1/3)
\]
Aber: $\pi_1 p_{12} = 1/3 \neq 0 = \pi_2 p_{21}$ -- Detailed Balance verletzt!
 \end{frame}

\begin{frame}
    \frametitle{Markov-Ketten}
\framesubtitle{Anwendung: Google PageRank}
    \begin{block}{Idee (Brin \& Page, 1998)}
Das Web ist ein gerichteter Graph. Ein ``Random Surfer'' klickt zufällig auf Links der aktuellen Seite $\Rightarrow$ Markov-Kette!
\end{block}
\begin{block}{Naives Modell}
Zustandsraum $S = \{\text{Webseiten}\}$, Übergangsmatrix:
\[
p_{ij} = \begin{cases} \frac{1}{\text{out}(i)} & \text{falls } i \to j \text{ (Link vorhanden)} \\ 0 & \text{sonst} \end{cases}
\]
mit $\text{out}(i)$ = Anzahl ausgehender Links von Seite $i$; $P$ ist \textbf{zeilenstochastisch}.
\end{block}
\textbf{Problem:} Nicht irreduzibel (Sackgassen) und evtl.\ periodisch!
 \end{frame}


%% =============================================
%% PAGERANK: GOOGLE-MATRIX
%% =============================================

\begin{frame}
    \frametitle{Markov-Ketten}
\framesubtitle{Google PageRank -- Die Google-Matrix}
    \begin{block}{Lösung: Dämpfungsfaktor $\alpha \in (0,1)$}
\[
G = \alpha \cdot P + (1 - \alpha) \cdot \frac{1}{|S|} \, \mathbb{I}
\]
mit $|S|$ = Anzahl aller Seiten in $S$; auch $G$ ist \textbf{zeilenstochastisch}.\\
Mit Wahrscheinlichkeit $\alpha$: folge einem Link.\\
Mit Wahrscheinlichkeit $1-\alpha$: springe zu einer zufälligen Seite.
\end{block}
\begin{block}{Eigenschaften von $G$ (typisch: $\alpha = 0.85$)}
\begin{itemize}
\item Alle Einträge $> 0$ $\Rightarrow$ \textbf{irreduzibel} und \textbf{aperiodisch}.
\item Hauptsatz anwendbar: $\exists!$ invariante Verteilung $\pi$.
\item $\pi_i$ = \textbf{PageRank} der Seite $i$ (Wichtigkeit).
\end{itemize}
\end{block}
 \end{frame}

%% KARTENMISCHEN: MODELL
%% =============================================

\begin{frame}
    \frametitle{Markov-Ketten}
\framesubtitle{Anwendung: Kartenmischen}
    \begin{block}{Modell}
Ein Kartendeck mit $n$ Karten = eine Permutation $\sigma \in S_n$.\\[0.2cm]
Zustandsraum: $S = S_n$ (alle $n!$ Permutationen).\\[0.1cm]
Ein Mischvorgang $\hat{=}$ ein Übergang $\sigma \to \sigma'$.
\end{block}
\begin{block}{Beispiel: Top-to-Random Shuffle}
Nimm die oberste Karte und stecke sie an eine zufällige Position:
\[
p_{\sigma, \sigma'} = \begin{cases} \frac{1}{n} & \text{falls } \sigma' \text{ durch Einfügen der Top-Karte entsteht} \\ 0 & \text{sonst} \end{cases}
\]
\end{block}
\textbf{Markov-Eigenschaft:} Die nächste Anordnung hängt nur von der aktuellen ab!
 \end{frame}


%% =============================================
%% KARTENMISCHEN: ANALYSE
%% =============================================

\begin{frame}
    \frametitle{Markov-Ketten}
\framesubtitle{Kartenmischen -- Analyse}
    \begin{block}{Eigenschaften der Misch-Kette}
\begin{itemize}
\item \textbf{Irreduzibel:} Jede Permutation ist von jeder anderen erreichbar.
\item \textbf{Aperiodisch:} Top-Karte kann an gleicher Stelle bleiben ($p_{\sigma\sigma} = 1/n > 0$).
\item $\Rightarrow$ Hauptsatz anwendbar!
\end{itemize}
\end{block}
\begin{block}{Invariante Verteilung}
Die Gleichverteilung
\[
\pi = \left(\frac{1}{n!}, \ldots, \frac{1}{n!}\right)
\]
ist invariant; wegen Irreduzibilit"at ist sie die \textbf{eindeutige} invariante Verteilung.
\end{block}
 \end{frame}


\begin{frame}
    \frametitle{Markov-Ketten}
\framesubtitle{Kartenmischen -- Warum ist $\pi$ invariant?}
    \begin{block}{Gleichverteilung auf $S_n$}
$\pi(\sigma)=\frac{1}{n!}$ für alle $\sigma \in S_n$ bedeutet: Jede Permutation ist gleich wahrscheinlich.
\end{block}
\begin{block}{Erster Schritt}
Fixiere eine Zielpermutation $\tau$.\\
Es gibt genau $n$ Vorgänger, die in einem Schritt nach $\tau$ führen: für jede Einfügeposition genau einen.\\
Jeder dieser Vorgänger hat unter der Gleichverteilung Wahrscheinlichkeit $\frac{1}{n!}$.
\end{block}
 \end{frame}


\begin{frame}
    \frametitle{Markov-Ketten}
\framesubtitle{Kartenmischen -- Warum ist $\pi$ invariant?}
\begin{block}{Zweiter Schritt}
Von jedem dieser $n$ Vorgänger geht man mit Wahrscheinlichkeit $\frac{1}{n}$ nach $\tau$.\\
Also ist die Gesamtwahrscheinlichkeit für $\tau$ nach einem Schritt
\[
n \cdot \frac{1}{n!}\cdot \frac{1}{n} = \frac{1}{n!} = \pi(\tau).
\]
Also bleibt die Gleichverteilung nach einem Mischschritt erhalten.
\end{block}
 \end{frame}


\begin{frame}
    \frametitle{Markov-Ketten}
\framesubtitle{Kartenmischen -- Warum eindeutig?}
\begin{block}{Warum eindeutig?}
Hier gilt:
\begin{itemize}
\item Der Zustandsraum $S_n$ ist endlich.
\item Die Kette ist irreduzibel.
\end{itemize}
Nach dem Hauptsatz gibt es deshalb genau eine invariante Verteilung.\\
Da die Gleichverteilung bereits invariant ist, muss sie diese eindeutige invariante Verteilung sein.
\end{block}
 \end{frame}


\begin{frame}
    \frametitle{Markov-Ketten}
\framesubtitle{Kartenmischen -- Warum $n\ln n$?}
    \begin{block}{Mischzeit (Mixing Time)}
Top-to-Random: $\approx n \ln n$ Schritte (für $n=52$: ca.\ $200$).
\end{block}
    \begin{block}{Heuristik (Coupon-Collector-Idee)}
Eine Karte ist erst dann wirklich ``durchmischt'', wenn sie irgendwann selbst oben lag und damit aktiv bewegt wurde.
\end{block}
\begin{block}{Erster Schritt}
Für eine feste Karte gilt nach $k$ Schritten:
\[
\left(1-\frac{1}{n}\right)^k \approx e^{-k/n}
\]
Sie wurde also mit ungefähr dieser Wahrscheinlichkeit noch nie bewegt.
\end{block}
 \end{frame}


\begin{frame}
    \frametitle{Markov-Ketten}
\framesubtitle{Kartenmischen -- Warum $n\ln n$?}
    \begin{block}{Zweiter Schritt}
Damit ist die erwartete Zahl noch nie bewegter Karten ungefähr
\[
n\,e^{-k/n}.
\]
``Gut gemischt'' ist das Deck erst, wenn diese Zahl klein wird, also etwa von Ordnung $1$:
\[
n\,e^{-k/n} \approx 1
\quad \Longrightarrow \quad
k \approx n\ln n.
\]
\end{block}
 \end{frame}

\framesubtitle{Zusammenfassung}

\begin{itemize}

\item \textbf{Stochastische Matrix:} Zeilensummen $= 1$, Einträge $\geq 0$.

\end{block}

%% IDEE MCMC
%% =============================================

\begin{frame}
    \frametitle{MCMC}
\framesubtitle{Idee: Sampling durch Markov-Ketten}
    \begin{block}{Grundidee}
Konstruiere eine Markov-Kette, deren \textbf{invariante Verteilung} genau die gewünschte Zielverteilung $\pi$ ist.
\end{block}
\begin{block}{Vorgehen}
\begin{enumerate}
\item Wähle Startzustand $X_0$ beliebig.
\item Simuliere die Kette: $X_0, X_1, X_2, \ldots$
\item Nach genügend Schritten (Burn-in): $X_n \sim \pi$ (approximativ).
\item Verwende $X_n, X_{n+1}, \ldots$ als Stichprobe aus $\pi$.
\end{enumerate}
\end{block}
\begin{block}{Warum funktioniert das?}
Hauptsatz: Irreduzibel + aperiodisch $\Rightarrow$ $X_n \to \pi$ in Verteilung!
\end{block}
 \end{frame}


%% =============================================
%% METROPOLIS-HASTINGS
%% =============================================

\begin{frame}
    \frametitle{MCMC}
\framesubtitle{Der Metropolis-Hastings-Algorithmus}
    \begin{block}{Gegeben}
\begin{itemize}
\item Zielverteilung $\pi$ (bis auf Normierung bekannt): $\pi(x) \propto f(x)$
\item Vorschlagsverteilung $q(y \mid x)$ (frei wählbar, leicht zu sampeln)
\end{itemize}
\end{block}
\begin{block}{Algorithmus (ein Schritt: $X_n = x \to X_{n+1} = ?$)}
\begin{enumerate}
\item Ziehe Vorschlag $y \sim q(\cdot \mid x)$.
\item Berechne Akzeptanzwahrscheinlichkeit (eine \textbf{Zahl} $\in [0,1]$):
\[
\alpha(x, y) = \min\!\left(1, \;\frac{\pi(y)\, q(x \mid y)}{\pi(x)\, q(y \mid x)}\right)
\]
\item Ziehe $U \sim \text{Uniform}(0,1)$.
\item Falls $U < \alpha(x,y)$: setze $X_{n+1} = y$ \quad (\textbf{akzeptiere}).\\
      Sonst: setze $X_{n+1} = x$ \quad (\textbf{ablehne}, bleibe stehen).
\end{enumerate}
\end{block}
 \end{frame}


%% =============================================
%% METROPOLIS-HASTINGS: ÜBERGANGSMATRIX
%% =============================================

\begin{frame}
    \frametitle{MCMC}
\framesubtitle{Metropolis-Hastings -- Die Übergangsmatrix $P$}
    \begin{block}{Was ist $P(x,y)$?}
Für $x \neq y$: Wahrscheinlichkeit, von $x$ nach $y$ zu springen.
\end{block}
\begin{block}{Definition der Übergangsmatrix}
\[
P(x,y) = q(y \mid x) \cdot \alpha(x,y) \qquad (x \neq y)
\]
mit $q(y \mid x)$ = Vorschlagswahrscheinlichkeit und $\alpha(x,y)$ = Akzeptanzwahrscheinlichkeit.
\end{block}
\begin{block}{Verbleibe-Wahrscheinlichkeit}
\[
P(x,x) = 1 - \sum_{y \neq x} P(x,y) \quad \text{(alles was übrig bleibt)}
\]
\end{block}
 \end{frame}


%% =============================================
%% METROPOLIS-HASTINGS: WARUM FUNKTIONIERT ES?
%% =============================================

\begin{frame}
    \frametitle{MCMC}
\framesubtitle{Metropolis-Hastings -- Korrektheit}
    \begin{block}{Satz: $\pi$ erfüllt Detailed Balance bzgl.\ $P$}
\[
\pi(x)\, P(x,y) = \pi(y)\, P(y,x) \quad \text{für alle } x \neq y
\]
\end{block}
\begin{block}{Warum?}
$\alpha$ ist genau so konstruiert, dass die ``Flüsse'' balanciert sind:\\[0.1cm]
$\alpha(x,y) = \min\!\left(1, \frac{\pi(y)q(x|y)}{\pi(x)q(y|x)}\right)$ korrigiert die Asymmetrie von $q$ und $\pi$.
\end{block}
 \end{frame}


%% =============================================
%% METROPOLIS-HASTINGS: BEWEIS
%% =============================================

\begin{frame}
    \frametitle{MCMC}
\framesubtitle{Metropolis-Hastings -- Beweis Detailed Balance}
    \begin{block}{Beweis}
O.B.d.A.\ $\pi(x)q(y|x) \geq \pi(y)q(x|y)$\; (sonst vertausche $x$ und $y$). Dann:
$\alpha(x,y) = \frac{\pi(y)\,q(x|y)}{\pi(x)\,q(y|x)} \leq 1$ \; und \; $\alpha(y,x) = 1$.
\end{block}
\textbf{Linke Seite:}
\[
\pi(x)\, P(x,y) = \pi(x)\, q(y|x) \cdot \frac{\pi(y)\,q(x|y)}{\pi(x)\,q(y|x)} = \pi(y)\,q(x|y)
\]
\textbf{Rechte Seite:}
\[
\pi(y)\, P(y,x) = \pi(y)\, q(x|y) \cdot \underbrace{\alpha(y,x)}_{= 1} = \pi(y)\,q(x|y)
\]
\begin{block}{Ergebnis}
$\pi(x) P(x,y) = \pi(y) q(x|y) = \pi(y) P(y,x) \quad \checkmark$ \quad $\Rightarrow$ $\pi$ ist invariant! \hfill $\square$
\end{block}
 \end{frame}


%% =============================================
%% METROPOLIS-HASTINGS: KONVERGENZ
%% =============================================

\begin{frame}
    \frametitle{MCMC}
\framesubtitle{Metropolis-Hastings -- Konvergenz}
    \begin{block}{Warum konvergiert MH zur Zielverteilung $\pi$?}
Drei Voraussetzungen des Hauptsatzes sind erfüllt:
\end{block}
\begin{enumerate}
\item \textbf{$\pi$ ist invariant:} Detailed Balance (gerade bewiesen).
\item \textbf{Irreduzibel:} Falls $q(y \mid x) > 0$ für alle $x,y$, dann:
\[
P(x,y) = q(y|x) \cdot \alpha(x,y) > 0 \quad \forall\, x \neq y
\]
Jeder Zustand ist in einem Schritt erreichbar.
\item \textbf{Aperiodisch:} Vorschlag kann abgelehnt werden $\Rightarrow$ $P(x,x) > 0$ (Selbstschleife) $\Rightarrow$ $d(x) = 1$.
\end{enumerate}
\begin{block}{Schlussfolgerung (Hauptsatz)}
\[
\text{Irreduzibel + aperiodisch} \;\Rightarrow\; P^n \to \pi \quad \text{(Konvergenz!)}
\]
Die Normierungskonstante von $\pi$ wird \textbf{nie} benötigt ($\alpha$ enthält nur Verhältnisse).
\end{block}
 \end{frame}


%% =============================================
%% SPEZIALFALL: METROPOLIS
%% =============================================

\begin{frame}
    \frametitle{MCMC}
\framesubtitle{Spezialfall: Metropolis-Algorithmus}
    \begin{block}{Symmetrischer Vorschlag: $q(y \mid x) = q(x \mid y)$}
Dann vereinfacht sich die Akzeptanzwahrscheinlichkeit zu:
\[
\alpha(x, y) = \min\!\left(1, \;\frac{\pi(y)}{\pi(x)}\right)
\]
\end{block}
\begin{block}{Interpretation}
\begin{itemize}
\item $\pi(y) \geq \pi(x)$: Vorschlag wird \textbf{immer} akzeptiert (bergauf).
\item $\pi(y) < \pi(x)$: Vorschlag wird mit W'keit $\pi(y)/\pi(x)$ akzeptiert (bergab).
\end{itemize}
$\Rightarrow$ Die Kette besucht häufiger Zustände mit hohem $\pi$-Wert!
\end{block}
\begin{block}{Typischer Vorschlag}
$q(y \mid x) = \mathcal{N}(x, \sigma^2)$ \quad (Random Walk Metropolis)
\end{block}
 \end{frame}


%% =============================================
%% MCMC INTEGRATION
%% =============================================

\begin{frame}
    \frametitle{MCMC}
\framesubtitle{Monte-Carlo-Integration}
    \begin{block}{Trick: künstliche Verteilung einführen}
Für
\[
I = \int h(x)\, dx
\]
wähle $\pi$ mit $\pi(x) > 0$ auf dem relevanten Bereich. Dann:
\[
I = \int \frac{h(x)}{\pi(x)}\, \pi(x)\, dx = \E_\pi\!\left[\frac{h(X)}{\pi(X)}\right]
\]
\end{block}
\begin{block}{Ergodentheorem}
\[
\frac{1}{N} \sum_{n=1}^{N} f(X_n) \;\xrightarrow{N \to \infty}\; \E_\pi[f] \quad \text{(fast sicher)}
\]
Hier: $f(x) = h(x)/\pi(x)$.
\end{block}
 \end{frame}


\begin{frame}
    \frametitle{MCMC}
\framesubtitle{Monte-Carlo-Integration in der Praxis}
    \begin{block}{Schätzer für ein allgemeines Integral}
Nach Burn-in $B$:
\[
\widehat I_N = \frac{1}{N-B} \sum_{n=B+1}^{N} \frac{h(X_n)}{\pi(X_n)}
\]
als Näherung für $I = \int h(x)\, dx$.
\end{block}
\begin{block}{Spezialfall}
Will man schon $\E_\pi[g] = \int g(x)\,\pi(x)\,dx$ berechnen, genügt einfach
\[
\frac{1}{N-B} \sum_{n=B+1}^{N} g(X_n)
\]
ohne den Faktor $1/\pi(X_n)$.
\end{block}
 \end{frame}


%% =============================================
%% MCMC: BEISPIEL
%% =============================================

\begin{frame}
    \frametitle{MCMC}
\framesubtitle{Beispiel: Normalverteilung sampeln}
    \begin{block}{Ziel}
Sample aus $\pi(x) \propto e^{-x^2/2}$ (Standardnormalverteilung).
\end{block}
\begin{block}{Random Walk Metropolis}
\begin{enumerate}
\item Start: $X_0 = 0$.
\item Vorschlag: $y = X_n + \varepsilon$, \quad $\varepsilon \sim \text{Uniform}(-\delta, \delta)$.
\item Akzeptanz: $\alpha = \min\!\left(1, \frac{e^{-y^2/2}}{e^{-X_n^2/2}}\right) = \min\!\left(1, e^{(X_n^2 - y^2)/2}\right)$.
\item $U \sim \text{Uniform}(0,1)$: Falls $U < \alpha$, setze $X_{n+1} = y$, sonst $X_{n+1} = X_n$.
\end{enumerate}
\end{block}
Nach Burn-in: Histogramm von $(X_n)$ $\approx$ Glockenkurve!
 \end{frame}

\begin{frame}
    \frametitle{MCMC}
\framesubtitle{Anwendungen}
    \begin{block}{Wo wird MCMC eingesetzt?}
\begin{itemize}
\item \textbf{Bayessche Statistik:} Sampling aus Posterior-Verteilungen.
\item \textbf{Statistische Physik:} Simulation von Ising-Modellen, Boltzmann-Verteilungen.
\item \textbf{Machine Learning:} Training von Boltzmann-Maschinen, Variational Inference.
\item \textbf{Kombinatorische Optimierung:} Simulated Annealing.
\item \textbf{Numerische Integration:} Hochdimensionale Integrale.
\end{itemize}
\end{block}
\begin{block}{Zusammenhang zu Markov-Ketten}
MCMC \textbf{konstruiert} eine Markov-Kette mit gewünschtem $\pi$.\\
Hauptsatz garantiert Konvergenz. Detailed Balance liefert Korrektheit.
\end{block}
 \end{frame}
